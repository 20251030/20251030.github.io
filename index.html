function drawPasswordBg() { pCanvas.width = window.innerWidth; pCanvas.height = window.innerHeight; var w = pCanvas.width; var h = pCanvas.height; // 深色背景 var grad = pCtx.createLinearGradient(0, 0, 0, h); grad.addColorStop(0, '#1a1a1a'); grad.addColorStop(1, '#0a0a0a'); pCtx.fillStyle = grad; pCtx.fillRect(0, 0, w, h); // ===== 从左下角斜着探出来的安卓机器人 ===== pCtx.save(); pCtx.globalAlpha = 0.12; // 透明度，越低越淡 pCtx.translate(-80, h + 80); // 移到左下角 pCtx.rotate(-Math.PI / 6); // 逆时针斜 30 度 pCtx.scale(6, 6); // 放大 6 倍 pCtx.strokeStyle = '#3ddc84'; pCtx.fillStyle = '#3ddc84'; pCtx.lineWidth = 3; // 头 pCtx.beginPath(); pCtx.arc(0, 0, 40, Math.PI, 0); pCtx.closePath(); pCtx.fill(); // 天线 pCtx.beginPath(); pCtx.moveTo(-20, -35); pCtx.lineTo(-30, -55); pCtx.stroke(); pCtx.beginPath(); pCtx.moveTo(20, -35); pCtx.lineTo(30, -55); pCtx.stroke(); // 眼睛 pCtx.fillStyle = '#111'; pCtx.beginPath(); pCtx.arc(-15, -15, 4, 0, Math.PI * 2); pCtx.fill(); pCtx.beginPath(); pCtx.arc(15, -15, 4, 0, Math.PI * 2); pCtx.fill(); // 身体（只画上半部分，够大就行） pCtx.fillStyle = '#3ddc84'; pCtx.fillRect(-40, 5, 80, 200); // 手臂 pCtx.fillRect(-60, 15, 15, 150); pCtx.fillRect(45, 15, 15, 150); pCtx.restore(); }
